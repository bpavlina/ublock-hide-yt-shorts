! Element and attribute selectors are plain CSS (uBO's :has() runs natively in the browser),
! so they apply before first paint and don't depend on the UI language.
! :has-text() is kept only where nothing else identifies the element.

! Shorts shelves (homepage, channel Home tab, watch page, Shorts remixes, subscriptions) except on history page
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)ytd-reel-shelf-renderer
www.youtube.com##ytd-rich-section-renderer:has(ytd-rich-shelf-renderer[is-shorts])

! Single Shorts in video lists and grids
www.youtube.com##ytd-video-renderer:has([overlay-style="SHORTS"], a[href^="/shorts/"])
www.youtube.com##ytd-rich-item-renderer:has(ytm-shorts-lockup-view-model, ytm-shorts-lockup-view-model-v2)

! Shorts shelves on search page
www.youtube.com##grid-shelf-view-model:has(a[href^="/shorts/"])

! Shorts filter chip on homepage and search page
www.youtube.com##yt-chip-cloud-chip-renderer:has-text(/^\s*Shorts\s*$/)

! Shorts button in sidebar (full and mini guide)
www.youtube.com##ytd-guide-entry-renderer:has(> a[title="Shorts"])
www.youtube.com##ytd-mini-guide-entry-renderer:has(> a[title="Shorts"])

! Shorts tab on channel pages
www.youtube.com##yt-tab-shape[tab-title="Shorts"]

! Shorts in the notifications area
www.youtube.com##ytd-notification-renderer:has(> a[href^="/shorts/"])
