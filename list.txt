! Most rules are plain CSS, which uBO hands to the browser. :has-text() is used only
! where nothing else identifies the element. Rules matching a label ("Shorts") include
! the translations YouTube uses in its interface languages.
! Content rules skip the History page, so watched Shorts stay listed there.

! Shorts shelves (homepage, channel Home tab, watch page, Shorts remixes, subscriptions) except on history page
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)ytd-reel-shelf-renderer
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)ytd-rich-section-renderer:has(ytd-rich-shelf-renderer[is-shorts])

! Single Shorts in video lists and grids
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)ytd-video-renderer:has([overlay-style="SHORTS"], a[href^="/shorts/"])
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)ytd-rich-item-renderer:has(ytm-shorts-lockup-view-model, ytm-shorts-lockup-view-model-v2)

! Shorts shelves on search page
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)grid-shelf-view-model:has(a[href^="/shorts/"])

! Single Shorts in mixed rows (e.g. a channel's "For You" shelf)
www.youtube.com##:matches-path(/^(?!\/feed\/history)/)yt-horizontal-list-renderer ytm-shorts-lockup-view-model-v2

! Single Shorts in the watch page's related videos
www.youtube.com##ytd-watch-next-secondary-results-renderer yt-lockup-view-model:has(a[href^="/shorts/"])

! Shorts filter chip on homepage and search page
www.youtube.com##yt-chip-cloud-chip-renderer:has-text(/^\s*(Shorts|Kortvideo\x27s|YouTube Shorts|Video Fupi|Kratki videoposnetki|Šortsi|Кыска видеолор|Шортси|کوته\u200cویدیوها|ショート)\s*$/)

! Shorts button in sidebar (full and mini guide)
www.youtube.com##ytd-guide-entry-renderer[is-primary]:has(> a:is([title="Shorts"],[title="Kortvideo's"],[title="YouTube Shorts"],[title="Video Fupi"],[title="Kratki videoposnetki"],[title="Šortsi"],[title="Кыска видеолор"],[title="Шортси"],[title="کوته‌ویدیوها"],[title="ショート"]))
www.youtube.com##ytd-mini-guide-entry-renderer:has(> a:is([title="Shorts"],[title="Kortvideo's"],[title="YouTube Shorts"],[title="Video Fupi"],[title="Kratki videoposnetki"],[title="Šortsi"],[title="Кыска видеолор"],[title="Шортси"],[title="کوته‌ویدیوها"],[title="ショート"]))

! Shorts tab on channel pages
www.youtube.com##yt-tab-shape:is([tab-title="Shorts"],[tab-title="Kortvideo's"],[tab-title="Curts"],[tab-title="Lühivideod"],[tab-title="YouTube Shorts"],[tab-title="Ama-Short"],[tab-title="Short"],[tab-title="Video Fupi"],[tab-title="Shorts-videoer"],[tab-title="Kratki videoposnetki"],[tab-title="Šorts"],[tab-title="Кыска видеолор"],[tab-title="Шортс"],[tab-title="Կարճ հոլովակներ"],[tab-title="کوته‌ویدیوها"],[tab-title="მოკლე ვიდეოები"],[tab-title="ショート"])

! Shorts in the notifications area
www.youtube.com##ytd-notification-renderer:has(> a[href^="/shorts/"])
